產業據點與區位#

  截至2025年12月花蓮都會區（指花蓮市與吉安鄉，亦包含新城鄉嘉禮村）尚在營業的製材廠與建材行約有19間，在空間選址上多集中在里漏部落附近的工業區與田浦車站一帶。標記年份為營業登記年，並非是確切的開業時間（年份資料來源：台灣公司網）。

def year_category(year):
    if year is None:
        return "無年份"
    elif year < 1990:
        return "1990年以前"
    else:
        return "1990年（含）以後"
category_colors = {
    "1990年以前": "green",
    "1990年（含）以後": "red",
    "無年份": "gray"
}
m = folium.Map(
    location=hualien_center,
    zoom_start=13,
    tiles="CartoDB positron"
)

for shop in wood_shops:
    year = shop.get("year")
    category = year_category(year)
    color = category_colors[category]

    popup_text = shop["name"]
    popup_text += f"<br>營業登記年：{year}" if year else "<br>營業登記年：未知"

    folium.Marker(
        location=[shop["lat"], shop["lon"]],
        popup=popup_text,
        icon=folium.Icon(color=color, icon="tree", prefix="fa")
    ).add_to(m)
